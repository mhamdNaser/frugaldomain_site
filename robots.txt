# https://frugaldomain.site

# The SPA fallback in .htaccess means an unknown path returns index.html with
# HTTP 200 rather than a 404, so anything that should not be indexed has to be
# named here as well as carrying robots=noindex in its own markup.

User-agent: *
Allow: /

Disallow: /admin
Disallow: /admin/
Disallow: /login
Disallow: /register
# A signed-in page. It renders nothing for a crawler and exists once per user.
Disallow: /my-library
# The account area: one private page per person, and a sign-in redirect for
# anyone else. The Arabic twins of all of these are listed one by one, never
# as a bare /ar prefix, which would hide the whole Arabic site.
Disallow: /account
Disallow: /ar/account
Disallow: /ar/login
Disallow: /ar/register
Disallow: /ar/my-library
# Query strings on the gallery produce endless near-duplicate URLs.
Disallow: /*?search=
Disallow: /*?page=

# The demos under /apps/<slug>/ are deliberately left crawlable. Each one
# carries robots=noindex in its own <head>, and a crawler has to be able to
# fetch the page to read that. Disallowing them here would hide the noindex
# and the URLs could still surface from links elsewhere.

# Answer engines and AI crawlers are allowed deliberately: being quoted in an
# AI answer is the point of the structured data on these pages. Two kinds are
# listed - the indexing crawlers that build an assistant's search corpus, and
# the user-triggered fetchers that retrieve a page because someone asked about
# it in a chat. The second kind matters most here; it is what happens when a
# developer asks an assistant for a free SVG-to-JSX converter.

# OpenAI
User-agent: GPTBot
Allow: /

User-agent: OAI-SearchBot
Allow: /

User-agent: ChatGPT-User
Allow: /
